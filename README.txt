 الطريقة 1: نشر مجاني على Render.com (مُوصى به)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. ارفع هذا المجلد على GitHub:
   - سجّل في https://github.com
   - أنشئ Repository جديد
   - ارفع كل الملفات

2. سجّل في https://render.com (مجاني)

3. اضغط "New +" ثم "Blueprint"

4. اربطه بالـ Repository على GitHub

5. Render سيقرأ ملف render.yaml تلقائياً وينشئ:
   ✓ Web Service (السيرفر)
   ✓ PostgreSQL Database (قاعدة البيانات)

6. بعد الانتهاء ستحصل على رابط مثل:
   https://fueltrack.onrender.com

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 الطريقة 2: نشر على Railway.app
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. ارفع على GitHub (نفس الخطوات أعلاه)

2. سجّل في https://railway.app

3. "New Project" → "Deploy from GitHub"

4. أضف خدمة PostgreSQL من Dashboard

5. أضف متغير البيئة:
   DATABASE_URL = (انسخه من خدمة PostgreSQL)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
